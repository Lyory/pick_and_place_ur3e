cmake_minimum_required(VERSION 3.8)
project(ur3_llm_control)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
  add_compile_options(-Wall -Wextra -Wpedantic)
endif()
set(CMAKE_CXX_STANDARD 17)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(rosidl_default_generators REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(std_msgs REQUIRED)
find_package(moveit_ros_planning_interface REQUIRED)
find_package(moveit_msgs REQUIRED)
find_package(shape_msgs REQUIRED)
find_package(gazebo_dev REQUIRED)
find_package(gazebo_ros REQUIRED)
find_package(yaml-cpp REQUIRED)
rosidl_generate_interfaces(${PROJECT_NAME}
  "srv/ExecuteSkill.srv"
  "srv/SetAttachment.srv"
  "srv/GetWorldState.srv"
)
add_executable(robot_skills src/robot_skills.cpp)
ament_target_dependencies(robot_skills rclcpp geometry_msgs sensor_msgs moveit_ros_planning_interface moveit_msgs shape_msgs)
target_link_libraries(robot_skills yaml-cpp)
rosidl_target_interfaces(robot_skills ${PROJECT_NAME} "rosidl_typesupport_cpp")
add_library(ur3_virtual_attachment SHARED src/virtual_attachment.cpp)
ament_target_dependencies(ur3_virtual_attachment rclcpp gazebo_ros)
target_include_directories(ur3_virtual_attachment PRIVATE ${GAZEBO_INCLUDE_DIRS})
target_link_libraries(ur3_virtual_attachment ${GAZEBO_LIBRARIES})
rosidl_target_interfaces(ur3_virtual_attachment ${PROJECT_NAME} "rosidl_typesupport_cpp")
install(TARGETS robot_skills DESTINATION lib/${PROJECT_NAME})
install(TARGETS ur3_virtual_attachment LIBRARY DESTINATION lib)
install(PROGRAMS scripts/llm_planner.py scripts/task_validator.py scripts/skill_executor.py scripts/task_manager.py DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY launch config prompt worlds models DESTINATION share/${PROJECT_NAME})
ament_export_dependencies(rosidl_default_runtime)
ament_package()
